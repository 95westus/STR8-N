; EDU 31 patterns, continuous loop; Ctrl-C restores entry LEDs and stops.
; Doubled timing: ~82 ms/frame, ~20 seconds/pass at nominal 8 MHz.
; ASM 0.5 accepts these named constants directly.
; IRQ masked, decimal clear; 816 E=1, D/DBR/PBR=0.
; BRK 01 ready; C starts looping; Ctrl-C reaches BRK 02 after cleanup.
ORG APP_ORG
SEI
CLD
; Validate initialized RAM ABI before calling any entry.
LDX #ABI_LAST
abi: LDA ABI_DESC,X
CMP signature,X
BNE bad
DEX
BPL abi
JMP start
bad: JMP bad
signature: BYTE B52,B41,B01,B0D
start: STZ frame
; Current rc1 discovery descriptor, not a public LED ABI.
LDA EDU_SIG0
CMP #DISC_S
BNE absent
LDA EDU_SIG1
CMP #DISC_V
BNE absent
LDA EDU_FORMAT
CMP #DISC_VERSION
BNE absent
LDA EDU_PRESENT
BEQ absent
; Do not select DDR or rewrite CRA/DDRA.
LDA LED_CRA
AND #CRA_DATA_BIT
BEQ absent
LDA LED_DATA
STA entry
BRA ready
absent: LDA #NO_EDU_CHAR
JSR ABI_PUTC
JSR ABI_NEWLINE
JMP ABI_HOLD
ready: BRK STOP_READY
NOP
next: JSR ABI_CANCEL
BCS finish
LDA frame
AND #FRAME_MASK
BNE show
; Print one-based pattern number in hexadecimal.
LDA frame
LSR A
LSR A
LSR A
INC A
JSR ABI_HEX
JSR ABI_NEWLINE
show: LDX frame
LDA patterns,X
STA LED_DATA
JSR delay
BCS finish
INC frame
LDA frame
CMP #FRAME_COUNT
BNE next
STZ frame
JMP next
finish: NOP
LDA entry
STA LED_DATA
done: BRK STOP_DONE
JMP ABI_HOLD
 ; Poll cancellation between dwell chunks; change DWELL_CHUNKS above.
delay: LDA #DWELL_CHUNKS
STA ticks
dwell: LDX #DWELL_OUTER
outer: LDY #LOOP_WRAP
inner: DEY
BNE inner
DEX
BNE outer
JSR ABI_CANCEL
BCS interrupted
DEC ticks
BNE dwell
CLC
RTS
interrupted: SEC
RTS
