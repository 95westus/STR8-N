; ASM 0.5 source document; paste unnumbered lines with PASTE.
; IRQ masked, decimal clear; 816 E=1, D/DBR/PBR=0.
; BRK 01 ready; BRK 02 complete; C continues; Ctrl-C cleans up.
ORG APP_ORG
SEI
CLD
; Validate initialized RAM ABI before calling any entry.
LDX #ABI_LAST
abi: LDA ABI_DESC,X
CMP signature,X
BNE bad
DEX
BPL abi
JMP start
bad: JMP bad
signature: BYTE B52,B41,B01,B0D
start: STZ frame
ready: BRK STOP_READY
NOP
next: JSR ABI_CANCEL
BCS finish
LDA frame
AND #FRAME_MASK
BNE show
; Print one-based pattern number in hexadecimal.
LDA frame
LSR A
LSR A
LSR A
INC A
JSR ABI_HEX
JSR ABI_NEWLINE
show: LDX frame
LDA patterns,X
STA mask
JSR burst
INC frame
LDA frame
CMP #FRAME_COUNT
BNE next
finish: NOP
done: BRK STOP_DONE
JMP ABI_HOLD
; Reads assert CS. No writes to these four windows.
; Owner confirms no hardware attached to CS0-CS3 windows.
burst: LDX #BURST_BATCHES
batch: LDY #LOOP_WRAP
pulse: LDA mask
AND #B01
BEQ skip0
LDA CS0
skip0: LDA mask
AND #B02
BEQ skip1
LDA CS1
skip1: LDA mask
AND #CRA_DATA_BIT
BEQ skip2
LDA CS2
skip2: LDA mask
AND #B08
BEQ skip3
LDA CS3
skip3: DEY
BNE pulse
DEX
BNE batch
RTS
patterns:
; 01: forward chase
BYTE B01,B02,B04,B08,B01,B02,B04,B08
; 02: reverse chase
BYTE B08,B04,B02,B01,B08,B04,B02,B01
; 03: bounce
BYTE B01,B02,B04,B08,B04,B02,B01,B00
; 04: alternating pairs
BYTE B05,B0A,B05,B0A,B05,B0A,B05,B0A
; 05: fill and drain
BYTE B01,B03,B07,B0F,B07,B03,B01,B00
; 06: outside and inside
BYTE B09,B06,B09,B06,B09,B06,B09,B06
; 07: all blink
BYTE B0F,B00,B0F,B00,B0F,B00,B0F,B00
; 08: adjacent pairs
BYTE B03,B06,B0C,B09,B03,B06,B0C,B09
frame: BYTE B00
mask: BYTE B00
END
.
