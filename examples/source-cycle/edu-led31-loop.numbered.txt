1030 ; EDU 31 patterns, continuous loop; Ctrl-C restores entry LEDs and stops.
1040 ; Doubled timing: ~82 ms/frame, ~20 seconds/pass at nominal 8 MHz.
1050 ; ASM 0.5 accepts these named constants directly.
1060 ; IRQ masked, decimal clear; 816 E=1, D/DBR/PBR=0.
1070 ; BRK 01 ready; C starts looping; Ctrl-C reaches BRK 02 after cleanup.
1080 ORG APP_ORG
1090 SEI
1100 CLD
1110 ; Validate initialized RAM ABI before calling any entry.
1120 LDX #ABI_LAST
1130 abi: LDA ABI_DESC,X
1140 CMP signature,X
1150 BNE bad
1160 DEX
1170 BPL abi
1180 JMP start
1190 bad: JMP bad
1200 signature: BYTE B52,B41,B01,B0D
1210 start: STZ frame
1220 ; Current rc1 discovery descriptor, not a public LED ABI.
1230 LDA EDU_SIG0
1240 CMP #DISC_S
1250 BNE absent
1260 LDA EDU_SIG1
1270 CMP #DISC_V
1280 BNE absent
1290 LDA EDU_FORMAT
1300 CMP #DISC_VERSION
1310 BNE absent
1320 LDA EDU_PRESENT
1330 BEQ absent
1340 ; Do not select DDR or rewrite CRA/DDRA.
1350 LDA LED_CRA
1360 AND #CRA_DATA_BIT
1370 BEQ absent
1380 LDA LED_DATA
1390 STA entry
1400 BRA ready
1410 absent: LDA #NO_EDU_CHAR
1420 JSR ABI_PUTC
1430 JSR ABI_NEWLINE
1440 JMP ABI_HOLD
1450 ready: BRK STOP_READY
1460 NOP
1470 next: JSR ABI_CANCEL
1480 BCS finish
1490 LDA frame
1500 AND #FRAME_MASK
1510 BNE show

1640 BCS finish
